34// A value type that references a SOS2 constraint from ModelStorage.
35// Usually this type is passed by copy.
36//
37// This type implements https://abseil.io/docs/cpp/guides/hash.
38class Sos2Constraint final : public ModelStorageItem {
39 public:
40 // The typed integer used for ids.
41 using IdType = Sos2ConstraintId;
42
43 inline Sos2Constraint(ModelStorageCPtr storage, Sos2ConstraintId id);
44
45 inline int64_t id() const;
46
47 inline Sos2ConstraintId typed_id() const;
48
49 inline int64_t num_expressions() const;
50 LinearExpression Expression(int index) const;
51 inline bool has_weights() const;
52 inline double weight(int index) const;
53 inline absl::string_view name() const;
54
55 // All variables that appear in the SOS2 constraint with a nonzero coefficient
56 // in any of the expressions. Order is not defined.
57 inline std::vector<Variable> NonzeroVariables() const;
58
59 // Returns a detailed string description of the contents of the constraint
60 // (not its name, use `<<` for that instead).
61 inline std::string ToString() const;
62
63 friend inline bool operator==(const Sos2Constraint& lhs,
64 const Sos2Constraint& rhs);
65 friend inline bool operator!=(const Sos2Constraint& lhs,
66 const Sos2Constraint& rhs);

69 friend std::ostream& operator<<(std::ostream& ostr,
70 const Sos2Constraint& constraint);
71
72 private:
73 Sos2ConstraintId id_;
74};

108std::vector<Variable> Sos2Constraint::NonzeroVariables() const {

112bool operator==(const Sos2Constraint& lhs, const Sos2Constraint& rhs) {
113 return lhs.id_ == rhs.id_ && lhs.storage() == rhs.storage();
114}
115
116bool operator!=(const Sos2Constraint& lhs, const Sos2Constraint& rhs) {
117 return !(lhs == rhs);
118}

120template <typename H>
121H AbslHashValue(H h, const Sos2Constraint& constraint) {
122 return H::combine(std::move(h), constraint.id_.value(), constraint.storage());
123}
