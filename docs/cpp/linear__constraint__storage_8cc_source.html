36LinearConstraintId LinearConstraintStorage::Add(const double lower_bound,
37 const double upper_bound,
38 const absl::string_view name) {
39 const LinearConstraintId id = next_id_;
40 ++next_id_;
41 Data& lin_con_data = linear_constraints_[id];
42 lin_con_data.lower_bound = lower_bound;
43 lin_con_data.upper_bound = upper_bound;
44 lin_con_data.name = std::string(name);
45 return id;
46}

64std::vector<LinearConstraintId> LinearConstraintStorage::ConstraintsFrom(
65 const LinearConstraintId start) const {
66 std::vector<LinearConstraintId> result;
67 for (const LinearConstraintId c :

76std::pair<LinearConstraintsProto, SparseDoubleMatrixProto>
77LinearConstraintStorage::Proto() const {
78 LinearConstraintsProto constraints;
79 const std::vector<LinearConstraintId> sorted_constraints =
80 SortedLinearConstraints();
81 for (const LinearConstraintId id : sorted_constraints) {
82 AppendConstraint(id, &constraints);
83 }
84 return {constraints, matrix_.Proto()};
85}

87void LinearConstraintStorage::AppendConstraint(
88 const LinearConstraintId constraint,
89 LinearConstraintsProto* const proto) const {
90 proto->add_ids(constraint.value());
91 const Data& data = linear_constraints_.at(constraint);
92 proto->add_lower_bounds(data.lower_bound);
93 proto->add_upper_bounds(data.upper_bound);
94 // TODO(b/238115672): we should potentially not fill this in on empty names.
95 proto->add_names(data.name);
96}

98LinearConstraintsProto LinearConstraintStorage::Proto(
99 const LinearConstraintId start, const LinearConstraintId end) const {
100 LinearConstraintsProto result;
101 for (const LinearConstraintId id :
102 util_intops::MakeStrongIntRange(start, end)) {
103 if (linear_constraints_.contains(id)) {
104 AppendConstraint(id, &result);
105 }
106 }
107 return result;
108}

114void LinearConstraintStorage::AdvanceCheckpointInDiff(
115 const VariableId variable_checkpoint, Diff& diff) const {
116 diff.checkpoint = std::max(diff.checkpoint, next_id_);
117 diff.variable_checkpoint =
118 std::max(diff.variable_checkpoint, variable_checkpoint);
119 diff.deleted.clear();
120 diff.lower_bounds.clear();
121 diff.upper_bounds.clear();
122 diff.matrix_keys.clear();
123}

125LinearConstraintStorage::UpdateResult LinearConstraintStorage::Update(
126 const Diff& diff, const absl::flat_hash_set<VariableId>& deleted_variables,
127 absl::Span<const VariableId> new_variables) const {
128 UpdateResult result;
129 for (const LinearConstraintId c : diff.deleted) {
130 result.deleted.Add(c.value());
131 }
132 absl::c_sort(result.deleted);
133
134 for (const LinearConstraintId c : SortedSetElements(diff.lower_bounds)) {
135 result.updates.mutable_lower_bounds()->add_ids(c.value());
136 result.updates.mutable_lower_bounds()->add_values(lower_bound(c));
137 }
138
139 for (const LinearConstraintId c : SortedSetElements(diff.upper_bounds)) {
140 result.updates.mutable_upper_bounds()->add_ids(c.value());
141 result.updates.mutable_upper_bounds()->add_values(upper_bound(c));
142 }
143
144 result.creates = Proto(diff.checkpoint, next_id_);
145
146 result.matrix_updates =
147 matrix_.Update(diff.deleted, ConstraintsFrom(diff.checkpoint),
148 deleted_variables, new_variables, diff.matrix_keys);
149 return result;
150}
