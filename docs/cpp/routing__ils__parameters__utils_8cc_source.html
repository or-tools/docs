24RecreateParameters::ParametersCase GetParameterCaseForRecreateHeuristic(
25 FirstSolutionStrategy::Value recreate_heuristic) {
26 switch (recreate_heuristic) {
27 case FirstSolutionStrategy::LOCAL_CHEAPEST_INSERTION:
28 return RecreateParameters::kLocalCheapestInsertion;
29 case FirstSolutionStrategy::LOCAL_CHEAPEST_COST_INSERTION:
30 return RecreateParameters::kLocalCheapestInsertion;
31 case FirstSolutionStrategy::SAVINGS:
32 return RecreateParameters::kSavings;
33 case FirstSolutionStrategy::PARALLEL_SAVINGS:
34 return RecreateParameters::kSavings;
35 case FirstSolutionStrategy::SEQUENTIAL_CHEAPEST_INSERTION:
36 return RecreateParameters::kGlobalCheapestInsertion;
37 case FirstSolutionStrategy::PARALLEL_CHEAPEST_INSERTION:
38 return RecreateParameters::kGlobalCheapestInsertion;
39 default:
40 return RecreateParameters::PARAMETERS_NOT_SET;
41 }
42}
